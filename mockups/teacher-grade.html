# 注意：n=0 时返回 0，n=1 时返回 1
def fib(n):
    if n == 0:
        return 0
    elif n == 1:
        return 1
    # 迭代法
    a, b = 0, 1
    for i in range(2, n + 1):
        a, b = b, a + b
    return b


n = int(input())
print(fib(n))
